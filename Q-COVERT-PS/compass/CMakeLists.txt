set(COMPASS_DIM 128 CACHE STRING "Compass benchmark embedding dimension")
set(COMPASS_DEGREE 128 CACHE STRING "Compass fixed neighbor slots (FAISS HNSW base degree is 2*M)")
set(COMPASS_PQ_SUBSPACES 8 CACHE STRING "Compass PQ code bytes (8-bit subquantizers)")
set(COMPASS_PQ_CENTROIDS 256 CACHE STRING "Compass centroids per PQ subspace")
set(COMPASS_SCALAR float32 CACHE STRING "Compass stored embedding: uint8 or float32")
if(COMPASS_SCALAR STREQUAL "uint8")
  set(COMPASS_SCALAR_ID 2)
elseif(COMPASS_SCALAR STREQUAL "float32")
  set(COMPASS_SCALAR_ID 1)
else()
  message(FATAL_ERROR "COMPASS_SCALAR must be uint8 or float32")
endif()
set(COMPASS_DEFINITIONS COMPASS_DIM=${COMPASS_DIM} COMPASS_DEGREE=${COMPASS_DEGREE}
    COMPASS_PQ_SUBSPACES=${COMPASS_PQ_SUBSPACES} COMPASS_PQ_CENTROIDS=${COMPASS_PQ_CENTROIDS}
    COMPASS_SCALAR_ID=${COMPASS_SCALAR_ID})
set(COMPASS_DEFINITIONS ${COMPASS_DEFINITIONS} PARENT_SCOPE)
add_library(compass_do INTERFACE)
target_include_directories(compass_do INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(compass_do INTERFACE RingDORAM::qcovert)
target_compile_definitions(compass_do INTERFACE ${COMPASS_DEFINITIONS})
if(RINGDORAM_BUILD_EXAMPLES)
  add_executable(compass_benchmark native.cpp)
  target_link_libraries(compass_benchmark PRIVATE compass_do)
endif()
if(RINGDORAM_BUILD_TESTS)
  add_executable(compass_test tests.cpp)
  target_link_libraries(compass_test PRIVATE compass_do)
  add_test(NAME compass.correctness COMMAND compass_test)
  if(RINGDORAM_ENABLE_SANITIZERS)
    set_tests_properties(compass.correctness PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
  endif()
  if(RINGDORAM_BENCHMARK_PYTHON AND RINGDORAM_BUILD_EXAMPLES)
    add_test(NAME compass.dataio COMMAND "${RINGDORAM_BENCHMARK_PYTHON}"
      "${CMAKE_CURRENT_SOURCE_DIR}/dataio_test.py")
    add_test(NAME compass.pipeline COMMAND "${RINGDORAM_BENCHMARK_PYTHON}"
      "${CMAKE_CURRENT_SOURCE_DIR}/pipeline_test.py" "${QCOVERT_PS_BINARY_DIR}")
    set_tests_properties(compass.pipeline PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1;ASAN_OPTIONS=detect_leaks=0" TIMEOUT 180)
  endif()
endif()
